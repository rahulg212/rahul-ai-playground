     4→     4→ * PlaygroundProvider — the single source of truth for the whole app.
     5→     5→ *
     6→     6→ * It owns:
     7→     7→ *   - conversations (persisted), the active conversation and its config

   155→   155→const PlaygroundContext = createContext<PlaygroundContextValue | null>(null);
   156→   156→
   157→   157→export function usePlayground(): PlaygroundContextValue {
   158→   158→  const context = useContext(PlaygroundContext);
   159→   159→  if (!context) {
   160→   160→    throw new Error('usePlayground must be used within a PlaygroundProvider');
   161→   161→  }
   162→   162→  return context;
   163→   163→}
   164→   164→
   165→   165→export function PlaygroundProvider({ children }: { children: React.ReactNode }) {
   166→   166→  /* ---------------- persisted state ---------------- */
   167→   167→  const {
   168→   168→    value: conversations,
   169→   169→    setValue: setConversations,
   170→   170→  } = useLocalStorage<Conversation[]>(STORAGE_KEYS.conversations, [], loadConversations);
   171→   171→  const { value: activeId, setValue: setActiveId } = useLocalStorage<string | null>(
   172→   172→    STORAGE_KEYS.activeConversationId,
   173→   173→    null,
   174→   174→  );
   175→   175→  const { value: settings, setValue: setSettings } = useLocalStorage<AppSettings>(
   176→   176→    STORAGE_KEYS.settings,
   177→   177→    defaultSettings(),
   178→   178→    loadSettings,
   179→   179→  );
   180→   180→  const { value: presets, setValue: setPresets } = useLocalStorage<Preset[]>(
   181→   181→    STORAGE_KEYS.presets,
   182→   182→    [],
   183→   183→    loadPresets,
   184→   184→  );
   185→   185→
   186→   186→  /* ---------------- in-memory state ---------------- */
   187→   187→  const [isStreaming, setIsStreaming] = useState(false);
   188→   188→  const [streamingConvId, setStreamingConvId] = useState<string | null>(null);
   189→   189→  const [streamingText, setStreamingText] = useState('');
   190→   190→  /** Live chain-of-thought for reasoning models (rendered in a Thinking block). */
   191→   191→  const [streamingReasoning, setStreamingReasoning] = useState('');
   192→   192→  const [chatError, setChatError] = useState<string | null>(null);
   193→   193→  const [rawDebug, setRawDebug] = useState<RawDebug | null>(null);
   194→   194→  const [testingKey, setTestingKey] = useState<ProviderId | null>(null);
   195→   195→  const [keyTestResults, setKeyTestResults] = useState<Partial<Record<ProviderId, TestKeyResult>>>({});
   196→   196→  const [models, setModels] = useState<Record<ProviderId, ModelsState>>(EMPTY_MODELS);
   197→   197→  const [sidebarOpen, setSidebarOpen] = useState(true);
   198→   198→  const [settingsOpen, setSettingsOpen] = useState(true);
   199→   199→  const [mobileDrawer, setMobileDrawer] = useState<'sidebar' | 'settings' | null>(null);
   200→   200→
   201→   201→  /* ---------------- refs for async flows ---------------- */
   202→   202→  const conversationsRef = useRef(conversations);
   203→   203→  const settingsRef = useRef(settings);
   204→   204→  const presetsRef = useRef(presets);
   205→   205→  const activeIdRef = useRef(activeId);
   206→   206→  const modelsRef = useRef(models);
   207→   207→  const abortRef = useRef<AbortController | null>(null);
   208→   208→  const streamingTextRef = useRef('');
   209→   209→  const streamingReasoningRef = useRef('');
   210→   210→  const isStreamingRef = useRef(false);
   211→   211→
   212→   212→  useEffect(() => { conversationsRef.current = conversations; }, [conversations]);
   213→   213→  useEffect(() => { settingsRef.current = settings; }, [settings]);
   214→   214→  useEffect(() => { presetsRef.current = presets; }, [presets]);
   215→   215→  useEffect(() => { activeIdRef.current = activeId; }, [activeId]);
   216→   216→
   217→   217→  const activeConversation = useMemo(() => {
   218→   218→    if (conversations.length === 0) return null;
   219→   219→    return conversations.find((c) => c.id === activeId) ?? conversations[0];
   220→   220→  }, [conversations, activeId]);
   221→   221→
   222→   222→  const activeConversationRef = useRef(activeConversation);
   223→   223→  useEffect(() => { activeConversationRef.current = activeConversation; }, [activeConversation]);
   224→   224→
   225→   225→  // Ensure there is always at least one conversation. Values load
   226→   226→  // synchronously from localStorage, so an empty list means a fresh browser.
   227→   227→  useEffect(() => {
   228→   228→    if (conversations.length === 0) {
   229→   229→      const fresh = createConversation();
   230→   230→      setConversations([fresh]);
   231→   231→      setActiveId(fresh.id);
   232→   232→    } else if (!conversations.some((c) => c.id === activeId)) {
   233→   233→      setActiveId(conversations[0].id);
   234→   234→    }
   235→   235→  }, [conversations, activeId, setConversations, setActiveId]);
   236→   236→
   237→   237→  const ready = activeConversation !== null;
   238→   238→
   239→   239→  /* ---------------- conversation mutations ---------------- */
   240→   240→
   241→   241→  const updateConversationById = useCallback(
   242→   242→    (
   243→   243→      id: string,
   244→   244→      patch: Partial<Conversation> | ((conversation: Conversation) => Partial<Conversation>),
   245→   245→    ) => {
   246→   246→      setConversations((previous) =>
   247→   247→        previous.map((conversation) =>
   248→   248→          conversation.id === id
   249→   249→            ? {
   250→   250→                ...conversation,
   251→   251→                ...(typeof patch === 'function' ? patch(conversation) : patch),
   252→   252→                updatedAt: Date.now(),
   253→   253→              }
   254→   254→            : conversation,
   255→   255→        ),
   256→   256→      );
   257→   257→    },
   258→   258→    [setConversations],
   259→   259→  );
   260→   260→
   261→   261→  const updateActive = useCallback(
   262→   262→    (patch: Partial<Conversation>) => {
   263→   263→      const id = activeConversationRef.current?.id;
   264→   264→      if (id) updateConversationById(id, patch);
   265→   265→    },
   266→   266→    [updateConversationById],
   267→   267→  );
   268→   268→
   269→   269→  const selectConversation = useCallback((id: string) => {
   270→   270→    setActiveId(id);
   271→   271→    setMobileDrawer(null);
   272→   272→  }, [setActiveId]);
   273→   273→
   274→   274→  const createNewChat = useCallback(() => {
   275→   275→    // New chats inherit the current provider/model/prompt/settings so the
   276→   276→    // user keeps testing without reconfiguration.
   277→   277→    const current = activeConversationRef.current;
   278→   278→    const fresh = createConversation(
   279→   279→      current?.provider ?? 'openai',
   280→   280→      current?.model,
   281→   281→      {
   282→   282→        systemPrompt: current?.systemPrompt ?? '',
   283→   283→        settings: current?.settings,
   284→   284→      },
   285→   285→    );
   286→   286→    setConversations((previous) => [fresh, ...previous]);
   287→   287→    setActiveId(fresh.id);
   288→   288→    setMobileDrawer(null);
   289→   289→    setChatError(null);
   290→   290→  }, [setConversations, setActiveId]);
   291→   291→
   292→   292→  const deleteConversation = useCallback(
   293→   293→    (id: string) => {
   294→   294→      const previous = conversationsRef.current;
   295→   295→      const remaining = previous.filter((c) => c.id !== id);
   296→   296→      if (remaining.length === 0) {
   297→   297→        const fresh = createConversation();
   298→   298→        setConversations([fresh]);
   299→   299→        setActiveId(fresh.id);
   300→   300→      } else {
   301→   301→        setConversations(remaining);
   302→   302→        if (id === activeIdRef.current) setActiveId(remaining[0].id);
   303→   303→      }
   304→   304→      toast.success('Conversation deleted.');
   305→   305→    },
   306→   306→    [setConversations, setActiveId],
   307→   307→  );
   308→   308→
   309→   309→  const renameConversation = useCallback(
   310→   310→    (id: string, title: string) => {
   311→   311→      updateConversationById(id, { title: title.trim() || 'Untitled' });
   312→   312→    },
   313→   313→    [updateConversationById],
   314→   314→  );
   315→   315→
   316→   316→  const clearActiveChat = useCallback(() => {
   317→   317→    updateActive({ messages: [] });
   318→   318→    setChatError(null);
   319→   319→    toast.success('Chat cleared.');
   320→   320→  }, [updateActive]);
   321→   321→
   322→   322→  /* ---------------- active conversation config ---------------- */
   323→   323→
   324→   324→  const setProvider = useCallback(
   325→   325→    (provider: ProviderId) => {
   326→   326→      // Switching provider keeps the chat but resets the model to a sensible
   327→   327→      // default (models differ across providers).
   328→   328→      updateActive({ provider, model: getAdapter(provider).defaultModel });
   329→   329→    },
   330→   330→    [updateActive],
   331→   331→  );
   332→   332→
   333→   333→  const setModel = useCallback((model: string) => updateActive({ model }), [updateActive]);
   334→   334→
   335→   335→  const setSystemPrompt = useCallback(
   336→   336→    (prompt: string) => updateActive({ systemPrompt: prompt }),
   337→   337→    [updateActive],
   338→   338→  );
   339→   339→
   340→   340→  const setGenerationSettings = useCallback(
   341→   341→    (patch: Partial<GenerationSettings>) => {
   342→   342→      updateActive((conversation) => ({
   343→   343→        settings: { ...conversation.settings, ...patch },
   344→   344→      }));
   345→   345→    },
   346→   346→    [updateActive],
   347→   347→  );
   348→   348→
   349→   349→  const resetGenerationSettings = useCallback(() => {
   350→   350→    updateActive({ settings: defaultGenerationSettings() });
   351→   351→    toast.success('Generation settings reset to defaults.');
   352→   352→  }, [updateActive]);
   353→   353→
   354→   354→  const setContextLimit = useCallback((limit: number) => updateActive({ contextLimit: limit }), [updateActive]);
   355→   355→
   356→   356→  /* ---------------- streaming ---------------- */
   357→   357→
   358→   358→  const runCompletion = useCallback(
   359→   359→    async (conversationId: string, history: ChatMessage[]) => {
   360→   360→      const conversation = conversationsRef.current.find((c) => c.id === conversationId);
   361→   361→      if (!conversation || history.length === 0) return;
   362→   362→
   363→   363→      const adapter = getAdapter(conversation.provider);
   364→   364→      const apiKey = settingsRef.current.apiKeys[conversation.provider];
   365→   365→      if (!apiKey) {
   366→   366→        setChatError(
   367→   367→          `No API key saved for ${providerLabel(conversation.provider)}. ` +
   368→   368→            'Open the settings panel on the right, paste your key and press Save.',
   369→   369→        );
   370→   370→        return;
   371→   371→      }
   372→   372→
   373→   373→      const limited = applyContextLimit(history, conversation.contextLimit);
   374→   374→
   375→   375→      const controller = new AbortController();
   376→   376→      abortRef.current = controller;
   377→   377→      isStreamingRef.current = true;
   378→   378→      streamingTextRef.current = '';
   379→   379→      streamingReasoningRef.current = '';
   380→   380→      setStreamingText('');
   381→   381→      setStreamingReasoning('');
   382→   382→      setStreamingConvId(conversationId);
   383→   383→      setIsStreaming(true);
   384→   384→      setChatError(null);
   385→   385→      setRawDebug(null);
   386→   386→
   387→   387→      const appendAssistant = (content: string, extra?: Partial<AssistantMeta>, reasoning?: string) => {
   388→   388→        const message: ChatMessage = {
   389→   389→          id: uid(),
   390→   390→          role: 'assistant',
   391→   391→          content,
   392→   392→          createdAt: Date.now(),
   393→   393→          reasoning: reasoning && reasoning.length > 0 ? reasoning : undefined,
   394→   394→          meta: {
   395→   395→            provider: conversation.provider,
   396→   396→            model: conversation.model,
   397→   397→            ...extra,
   398→   398→          },
   399→   399→        };
   400→   400→        updateConversationById(conversationId, (c) => ({
   401→   401→          messages: [...c.messages, message],
   402→   402→        }));
   403→   403→      };
   404→   404→
   405→   405→      try {
   406→   406→        // The adapter turns our internal format into the provider request and
   407→   407→        // parses the streamed response back — see src/lib/providerAdapters.
   408→   408→        const outcome = await runChatStream({
   409→   409→          adapter,
   410→   410→          apiKey,
   411→   411→          model: conversation.model || adapter.defaultModel,
   412→   412→          systemPrompt: conversation.systemPrompt,
   413→   413→          history: limited,
   414→   414→          settings: conversation.settings,
   415→   415→          signal: controller.signal,
   416→   416→          onDelta: (delta) => {
   417→   417→            streamingTextRef.current += delta;
   418→   418→            setStreamingText(streamingTextRef.current);
   419→   419→          },
   420→   420→          onReasoning: (delta) => {
   421→   421→            streamingReasoningRef.current += delta;
   422→   422→            setStreamingReasoning(streamingReasoningRef.current);
   423→   423→          },
   424→   424→        });
   425→   425→        appendAssistant(
   426→   426→          outcome.text || '*(empty response)*',
   427→   427→          {
   428→   428→            responseTimeMs: outcome.responseTimeMs,
   429→   429→            usage: outcome.usage,
   430→   430→          },
   431→   431→          outcome.reasoning,
   432→   432→        );
   433→   433→        setRawDebug({
   434→   434→          request: outcome.rawRequest,
   435→   435→          response: outcome.rawResponse,
   436→   436→          status: outcome.status,
   437→   437→          responseTimeMs: outcome.responseTimeMs,
   438→   438→        });
   439→   439→      } catch (error) {
   440→   440→        const playgroundError = toPlaygroundError(error);
   441→   441→        const partial = streamingTextRef.current;
   442→   442→        const partialReasoning = streamingReasoningRef.current;
   443→   443→        if (playgroundError.kind === 'abort') {
   444→   444→          // "Stop" pressed: keep whatever text already arrived.
   445→   445→          if (partial || partialReasoning) {
   446→   446→            appendAssistant(partial || '*(stopped before any text arrived)*', { stopped: true }, partialReasoning);
   447→   447→          }
   448→   448→          toast.info('Generation stopped.');
   449→   449→        } else {
   450→   450→          if (partial || partialReasoning) {
   451→   451→            appendAssistant(
   452→   452→              partial || '*(error before any text arrived)*',
   453→   453→              { stopped: true, error: playgroundError.message },
   454→   454→              partialReasoning,
   455→   455→            );
   456→   456→          }
   457→   457→          setChatError(playgroundError.message);
   458→   458→          toast.error(playgroundError.message);
   459→   459→          // Even failed attempts are useful in the Raw panel.
   460→   460→          if (playgroundError.rawRequest) {
   461→   461→            setRawDebug({
   462→   462→              request: playgroundError.rawRequest,
   463→   463→              response: `// Error: ${playgroundError.message}`,
   464→   464→              status: playgroundError.status,
   465→   465→            });
   466→   466→          }
   467→   467→        }
   468→   468→      } finally {
   469→   469→        abortRef.current = null;
   470→   470→        isStreamingRef.current = false;
   471→   471→        streamingTextRef.current = '';
   472→   472→        streamingReasoningRef.current = '';
   473→   473→        setIsStreaming(false);
   474→   474→        setStreamingConvId(null);
   475→   475→        setStreamingText('');
   476→   476→        setStreamingReasoning('');
   477→   477→      }
   478→   478→    },
   479→   479→    [updateConversationById],
   480→   480→  );
   481→   481→
   482→   482→  const sendMessage = useCallback(
   483→   483→    async (text: string) => {
   484→   484→      const conversation = activeConversationRef.current;
   485→   485→      if (!conversation || isStreamingRef.current) return;
   486→   486→      const userMessage: ChatMessage = {
   487→   487→        id: uid(),
   488→   488→        role: 'user',
   489→   489→        content: text,
   490→   490→        createdAt: Date.now(),
   491→   491→      };
   492→   492→      const history = [...conversation.messages, userMessage];
   493→   493→      // Auto-title the conversation from its first user message.
   494→   494→      const patch: Partial<Conversation> = { messages: history };
   495→   495→      if (conversation.messages.length === 0) patch.title = autoTitle(text);
   496→   496→      updateConversationById(conversation.id, patch);
   497→   497→      await runCompletion(conversation.id, history);
   498→   498→    },
   499→   499→    [runCompletion, updateConversationById],
   500→   500→  );
   501→   501→
   502→   502→  const stopStreaming = useCallback(() => {
   503→   503→    abortRef.current?.abort();
   504→   504→  }, []);
   505→   505→
   506→   506→  const editMessage = useCallback(
   507→   507→    async (messageId: string, newContent: string) => {
   508→   508→      const conversation = activeConversationRef.current;
   509→   509→      if (!conversation || isStreamingRef.current) return;
   510→   510→      const index = conversation.messages.findIndex((m) => m.id === messageId);
   511→   511→      if (index === -1) return;
   512→   512→      // Editing a message truncates everything after it, then re-asks the model.
   513→   513→      const history: ChatMessage[] = [
   514→   514→        ...conversation.messages.slice(0, index),
   515→   515→        { ...conversation.messages[index], content: newContent },
   516→   516→      ];
   517→   517→      updateConversationById(conversation.id, { messages: history });
   518→   518→      await runCompletion(conversation.id, history);
   519→   519→    },
   520→   520→    [runCompletion, updateConversationById],
   521→   521→  );
   522→   522→
   523→   523→  const deleteMessage = useCallback(
   524→   524→    (messageId: string) => {
   525→   525→      const conversation = activeConversationRef.current;
   526→   526→      if (!conversation) return;
   527→   527→      updateConversationById(conversation.id, (c) => ({
   528→   528→        messages: c.messages.filter((m) => m.id !== messageId),
   529→   529→      }));
   530→   530→    },
   531→   531→    [updateConversationById],
   532→   532→  );
   533→   533→
   534→   534→  const regenerate = useCallback(async () => {
   535→   535→    const conversation = activeConversationRef.current;
   536→   536→    if (!conversation || isStreamingRef.current) return;
   537→   537→    // Drop trailing assistant message(s) and ask again.
   538→   538→    let messages = conversation.messages;
   539→   539→    while (messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
   540→   540→      messages = messages.slice(0, -1);
   541→   541→    }
   542→   542→    if (messages.length === 0) return;
   543→   543→    updateConversationById(conversation.id, { messages });
   544→   544→    await runCompletion(conversation.id, messages);
   545→   545→  }, [runCompletion, updateConversationById]);
   546→   546→

   577→   577→    const cached = modelsRef.current[provider];

   579→   579→
   580→   580→    modelsRef.current = {
   581→   581→      ...modelsRef.current,
   582→   582→      [provider]: { ...cached, loading: true, error: null },
   583→   583→    };
   584→   584→    setModels(modelsRef.current);
   585→   585→
   586→   586→    try {
   587→   587→      const apiKey = settingsRef.current.apiKeys[provider];
   588→   588→      const list = await getAdapter(provider).fetchModels(apiKey);
   589→   589→      modelsRef.current = {
   590→   590→        ...modelsRef.current,
   591→   591→        [provider]: { loading: false, error: null, models: list },
   592→   592→      };
   593→   593→    } catch (error) {
   594→   594→      const playgroundError = toPlaygroundError(error);
   595→   595→      modelsRef.current = {
   596→   596→        ...modelsRef.current,
   597→   597→        [provider]: { loading: false, error: playgroundError.message, models: [] },
   598→   598→      };
   599→   599→    }
   600→   600→    setModels(modelsRef.current);
   601→   601→  }, []);
   602→   602→
   603→   603→  const testApiKey = useCallback(
   604→   604→    async (provider: ProviderId) => {
   605→   605→      const key = settingsRef.current.apiKeys[provider];
   606→   606→      if (!key) {
   607→   607→        setKeyTestResults((previous) => ({
   608→   608→          ...previous,
   609→   609→          [provider]: { ok: false, message: 'No key entered — type one first.' },
   610→   610→        }));
   611→   611→        return;
   612→   612→      }
   613→   613→      setTestingKey(provider);
   614→   614→      try {
   615→   615→        const result = await getAdapter(provider).testKey(key);
   616→   616→        setKeyTestResults((previous) => ({ ...previous, [provider]: result }));
   617→   617→        if (result.ok) {
   618→   618→          toast.success(result.message);
   619→   619→          // A valid key means the models endpoint is usable too — (re)load the
   620→   620→          // list right away so the Model section shows available models
   621→   621→          // immediately after a successful test.
   622→   622→          void loadModels(provider, true);
   623→   623→        } else {
   624→   624→          toast.error(result.message);
   625→   625→        }
   626→   626→      } catch (error) {
   627→   627→        const playgroundError = toPlaygroundError(error);
   628→   628→        setKeyTestResults((previous) => ({
   629→   629→          ...previous,
   630→   630→          [provider]: { ok: false, message: playgroundError.message },
   631→   631→        }));
   632→   632→        toast.error(playgroundError.message);
   633→   633→      } finally {
   634→   634→        setTestingKey(null);
   635→   635→      }
   636→   636→    },

   638→   638→  );

   651→   651→        if (existing) {
   652→   652→          toast.success(`Preset "${trimmed}" updated.`);
   653→   653→          return previous.map((p) => (p.id === existing.id ? { ...p, prompt } : p));
   654→   654→        }
   655→   655→        toast.success(`Preset "${trimmed}" saved.`);

   657→   657→      });
   658→   658→    },
