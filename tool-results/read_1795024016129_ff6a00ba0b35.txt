    82→    82→/**
    83→    83→ * Pick the best free model from a fetched list (OpenRouter only).
    84→    84→ * Prefers the app's default free model, then the first available one.
    85→    85→ * Returns null when the list has no free models (or is not loaded yet).
    86→    86→ */
    87→    87→function pickFreeModel(models: ModelInfo[]): string | null {
    88→    88→  const free = models.filter((m) => m.free);
    89→    89→  if (free.length === 0) return null;
    90→    90→  const preferred = free.find((m) => m.id === DEFAULT_MODELS.openrouter) ?? free[0];
    91→    91→  return preferred.id;
    92→    92→}

    98→    98→ * reliable instead of failing on the first rate-limit window.
    99→    99→ */

   183→   183→export function usePlayground(): PlaygroundContextValue {
   184→   184→  const context = useContext(PlaygroundContext);
   185→   185→  if (!context) {
   186→   186→    throw new Error('usePlayground must be used within a PlaygroundProvider');
   187→   187→  }
   188→   188→  return context;
   189→   189→}

   191→   191→export function PlaygroundProvider({ children }: { children: React.ReactNode }) {
   192→   192→  /* ---------------- persisted state ---------------- */
   193→   193→  const {
   194→   194→    value: conversations,
   195→   195→    setValue: setConversations,
   196→   196→  } = useLocalStorage<Conversation[]>(STORAGE_KEYS.conversations, [], loadConversations);
   197→   197→  const { value: activeId, setValue: setActiveId } = useLocalStorage<string | null>(
   198→   198→    STORAGE_KEYS.activeConversationId,
   199→   199→    null,
   200→   200→  );
   201→   201→  const { value: settings, setValue: setSettings } = useLocalStorage<AppSettings>(
   202→   202→    STORAGE_KEYS.settings,
   203→   203→    defaultSettings(),
   204→   204→    loadSettings,
   205→   205→  );
   206→   206→  const { value: presets, setValue: setPresets } = useLocalStorage<Preset[]>(
   207→   207→    STORAGE_KEYS.presets,
   208→   208→    [],
   209→   209→    loadPresets,
   210→   210→  );
   211→   211→
   212→   212→  /* ---------------- in-memory state ---------------- */
   213→   213→  const [isStreaming, setIsStreaming] = useState(false);
   214→   214→  const [streamingConvId, setStreamingConvId] = useState<string | null>(null);
   215→   215→  const [streamingText, setStreamingText] = useState('');
   216→   216→  /** Live chain-of-thought for reasoning models (rendered in a Thinking block). */
   217→   217→  const [streamingReasoning, setStreamingReasoning] = useState('');
   218→   218→  const [chatError, setChatError] = useState<string | null>(null);
   219→   219→  const [chatErrorAction, setChatErrorAction] = useState<ChatErrorAction | null>(null);
   220→   220→  const [rawDebug, setRawDebug] = useState<RawDebug | null>(null);
   221→   221→  const [testingKey, setTestingKey] = useState<ProviderId | null>(null);
   222→   222→  const [keyTestResults, setKeyTestResults] = useState<Partial<Record<ProviderId, TestKeyResult>>>({});
   223→   223→  const [models, setModels] = useState<Record<ProviderId, ModelsState>>(EMPTY_MODELS);
   224→   224→  const [sidebarOpen, setSidebarOpen] = useState(true);
   225→   225→  const [settingsOpen, setSettingsOpen] = useState(true);
   226→   226→  const [mobileDrawer, setMobileDrawer] = useState<'sidebar' | 'settings' | null>(null);
   227→   227→
   228→   228→  /* ---------------- refs for async flows ---------------- */
   229→   229→  const conversationsRef = useRef(conversations);
   230→   230→  const settingsRef = useRef(settings);
   231→   231→  const presetsRef = useRef(presets);
   232→   232→  const activeIdRef = useRef(activeId);
   233→   233→  const modelsRef = useRef(models);
   234→   234→  const abortRef = useRef<AbortController | null>(null);
   235→   235→  const streamingTextRef = useRef('');
   236→   236→  const streamingReasoningRef = useRef('');
   237→   237→  const isStreamingRef = useRef(false);
   238→   238→
   239→   239→  useEffect(() => { conversationsRef.current = conversations; }, [conversations]);
   240→   240→  useEffect(() => { settingsRef.current = settings; }, [settings]);
   241→   241→  useEffect(() => { presetsRef.current = presets; }, [presets]);
   242→   242→  useEffect(() => { activeIdRef.current = activeId; }, [activeId]);
   243→   243→
   244→   244→  const activeConversation = useMemo(() => {
   245→   245→    if (conversations.length === 0) return null;
   246→   246→    return conversations.find((c) => c.id === activeId) ?? conversations[0];
   247→   247→  }, [conversations, activeId]);
   248→   248→
   249→   249→  const activeConversationRef = useRef(activeConversation);
   250→   250→  useEffect(() => { activeConversationRef.current = activeConversation; }, [activeConversation]);
   251→   251→
   252→   252→  // Ensure there is always at least one conversation. Values load
   253→   253→  // synchronously from localStorage, so an empty list means a fresh browser.
   254→   254→  useEffect(() => {
   255→   255→    if (conversations.length === 0) {
   256→   256→      const fresh = createConversation();
   257→   257→      setConversations([fresh]);
   258→   258→      setActiveId(fresh.id);
   259→   259→    } else if (!conversations.some((c) => c.id === activeId)) {
   260→   260→      setActiveId(conversations[0].id);
   261→   261→    }
   262→   262→  }, [conversations, activeId, setConversations, setActiveId]);
   263→   263→
   264→   264→  // One-time migration of data written by older versions of the app
   265→   265→  // (e.g. raising maxTokens so reasoning models can think AND answer).
   266→   266→  useEffect(() => {
   267→   267→    setConversations((previous) => migrateConversations(previous));
   268→   268→  }, [setConversations]);
   269→   269→
   270→   270→  const ready = activeConversation !== null;
   271→   271→
   272→   272→  /* ---------------- conversation mutations ---------------- */
   273→   273→
   274→   274→  const updateConversationById = useCallback(
   275→   275→    (
   276→   276→      id: string,
   277→   277→      patch: Partial<Conversation> | ((conversation: Conversation) => Partial<Conversation>),
   278→   278→    ) => {
   279→   279→      setConversations((previous) =>
   280→   280→        previous.map((conversation) =>
   281→   281→          conversation.id === id
   282→   282→            ? {
   283→   283→                ...conversation,
   284→   284→                ...(typeof patch === 'function' ? patch(conversation) : patch),
   285→   285→                updatedAt: Date.now(),
   286→   286→              }
   287→   287→            : conversation,
   288→   288→        ),
   289→   289→      );
   290→   290→    },
   291→   291→    [setConversations],
   292→   292→  );
   293→   293→
   294→   294→  const updateActive = useCallback(
   295→   295→    (patch: Partial<Conversation>) => {
   296→   296→      const id = activeConversationRef.current?.id;
   297→   297→      if (id) updateConversationById(id, patch);
   298→   298→    },
   299→   299→    [updateConversationById],
   300→   300→  );
   301→   301→
   302→   302→  const selectConversation = useCallback((id: string) => {
   303→   303→    setActiveId(id);
   304→   304→    setMobileDrawer(null);
   305→   305→  }, [setActiveId]);
   306→   306→
   307→   307→  const createNewChat = useCallback(() => {
   308→   308→    // New chats inherit the current provider/model/prompt/settings so the
   309→   309→    // user keeps testing without reconfiguration.
   310→   310→    const current = activeConversationRef.current;
   311→   311→    const fresh = createConversation(
   312→   312→      current?.provider ?? 'openai',
   313→   313→      current?.model,
   314→   314→      {
   315→   315→        systemPrompt: current?.systemPrompt ?? '',
   316→   316→        settings: current?.settings,
   317→   317→      },
   318→   318→    );
   319→   319→    setConversations((previous) => [fresh, ...previous]);
   320→   320→    setActiveId(fresh.id);
   321→   321→    setMobileDrawer(null);
   322→   322→    setChatError(null);
   323→   323→    setChatErrorAction(null);
   324→   324→  }, [setConversations, setActiveId]);
   325→   325→
   326→   326→  const deleteConversation = useCallback(
   327→   327→    (id: string) => {
   328→   328→      const previous = conversationsRef.current;
   329→   329→      const remaining = previous.filter((c) => c.id !== id);
   330→   330→      if (remaining.length === 0) {
   331→   331→        const fresh = createConversation();
   332→   332→        setConversations([fresh]);
   333→   333→        setActiveId(fresh.id);
   334→   334→      } else {
   335→   335→        setConversations(remaining);
   336→   336→        if (id === activeIdRef.current) setActiveId(remaining[0].id);
   337→   337→      }
   338→   338→      toast.success('Conversation deleted.');
   339→   339→    },
   340→   340→    [setConversations, setActiveId],
   341→   341→  );
   342→   342→
   343→   343→  const renameConversation = useCallback(
   344→   344→    (id: string, title: string) => {
   345→   345→      updateConversationById(id, { title: title.trim() || 'Untitled' });
   346→   346→    },
   347→   347→    [updateConversationById],
   348→   348→  );
   349→   349→
   350→   350→  const clearActiveChat = useCallback(() => {
   351→   351→    updateActive({ messages: [] });
   352→   352→    setChatError(null);
   353→   353→    setChatErrorAction(null);
   354→   354→    toast.success('Chat cleared.');
   355→   355→  }, [updateActive]);
   356→   356→
   357→   357→  /* ---------------- active conversation config ---------------- */
   358→   358→
   359→   359→  const setProvider = useCallback(
   360→   360→    (provider: ProviderId) => {
   361→   361→      // Switching provider keeps the chat but resets the model to a sensible
   362→   362→      // default (models differ across providers).
   363→   363→      updateActive({ provider, model: getAdapter(provider).defaultModel });
   364→   364→    },
   365→   365→    [updateActive],
   366→   366→  );
   367→   367→
   368→   368→  const setModel = useCallback((model: string) => updateActive({ model }), [updateActive]);
   369→   369→
   370→   370→  const setSystemPrompt = useCallback(
   371→   371→    (prompt: string) => updateActive({ systemPrompt: prompt }),
   372→   372→    [updateActive],
   373→   373→  );
   374→   374→
   375→   375→  const setGenerationSettings = useCallback(
   376→   376→    (patch: Partial<GenerationSettings>) => {
   377→   377→      updateActive((conversation) => ({
   378→   378→        settings: { ...conversation.settings, ...patch },
   379→   379→      }));
   380→   380→    },
   381→   381→    [updateActive],
   382→   382→  );
   383→   383→
   384→   384→  const resetGenerationSettings = useCallback(() => {
   385→   385→    updateActive({ settings: defaultGenerationSettings() });
   386→   386→    toast.success('Generation settings reset to defaults.');
   387→   387→  }, [updateActive]);
   388→   388→
   389→   389→  const setContextLimit = useCallback((limit: number) => updateActive({ contextLimit: limit }), [updateActive]);
   390→   390→
   391→   391→  /* ---------------- streaming ---------------- */
   392→   392→
   393→   393→  const runCompletion = useCallback(
   394→   394→    async (conversationId: string, history: ChatMessage[]) => {
   395→   395→      const conversation = conversationsRef.current.find((c) => c.id === conversationId);
   396→   396→      if (!conversation || history.length === 0) return;
   397→   397→
   398→   398→      const adapter = getAdapter(conversation.provider);
   399→   399→      const apiKey = settingsRef.current.apiKeys[conversation.provider];
   400→   400→      if (!apiKey) {
   401→   401→        setChatError(
   402→   402→          `No API key saved for ${providerLabel(conversation.provider)}. ` +
   403→   403→            'Open the settings panel on the right, paste your key and press Save.',
   404→   404→        );
   405→   405→        return;
   406→   406→      }
   407→   407→
   408→   408→      const limited = applyContextLimit(history, conversation.contextLimit);
   409→   409→
   410→   410→      const controller = new AbortController();
   411→   411→      abortRef.current = controller;
   412→   412→      isStreamingRef.current = true;
   413→   413→      streamingTextRef.current = '';
   414→   414→      streamingReasoningRef.current = '';
   415→   415→      setStreamingText('');
   416→   416→      setStreamingReasoning('');
   417→   417→      setStreamingConvId(conversationId);
   418→   418→      setIsStreaming(true);
   419→   419→      setChatError(null);
   420→   420→      setChatErrorAction(null);
   421→   421→      setRawDebug(null);
   422→   422→
   423→   423→      const appendAssistant = (content: string, extra?: Partial<AssistantMeta>, reasoning?: string) => {
   424→   424→        const message: ChatMessage = {
   425→   425→          id: uid(),
   426→   426→          role: 'assistant',
   427→   427→          content,
   428→   428→          createdAt: Date.now(),
   429→   429→          reasoning: reasoning && reasoning.length > 0 ? reasoning : undefined,
   430→   430→          meta: {
   431→   431→            provider: conversation.provider,
   432→   432→            model: conversation.model,
   433→   433→            ...extra,
   434→   434→          },
   435→   435→        };
   436→   436→        updateConversationById(conversationId, (c) => ({
   437→   437→          messages: [...c.messages, message],
   438→   438→        }));
   439→   439→      };
   440→   440→
   441→   441→      try {
   442→   442→        // The adapter turns our internal format into the provider request and
   443→   443→        // parses the streamed response back — see src/lib/providerAdapters.
   444→   444→        // Transient provider failures (429 / 5xx) are retried automatically
   445→   445→        // as long as NOTHING has streamed yet — otherwise the partial text
   446→   446→        // would be duplicated in the conversation.
   447→   447→        let outcome: StreamOutcome | null = null;
   448→   448→        for (let attempt = 0; attempt <= MAX_AUTO_RETRIES; attempt += 1) {
   449→   449→          try {
   450→   450→            outcome = await runChatStream({
   451→   451→              adapter,
   452→   452→              apiKey,
   453→   453→              model: conversation.model || adapter.defaultModel,
   454→   454→              systemPrompt: conversation.systemPrompt,
   455→   455→              history: limited,
   456→   456→              settings: conversation.settings,
   457→   457→              signal: controller.signal,
   458→   458→              onDelta: (delta) => {
   459→   459→                streamingTextRef.current += delta;
   460→   460→                setStreamingText(streamingTextRef.current);
   461→   461→              },
   462→   462→              onReasoning: (delta) => {
   463→   463→                streamingReasoningRef.current += delta;
   464→   464→                setStreamingReasoning(streamingReasoningRef.current);
   465→   465→              },
   466→   466→            });
   467→   467→            break;
   468→   468→          } catch (retryError) {
   469→   469→            const retryPlaygroundError = toPlaygroundError(retryError, adapter.id);
   470→   470→            const nothingStreamed =
   471→   471→              streamingTextRef.current === '' && streamingReasoningRef.current === '';
   472→   472→            const transient =
   473→   473→              retryPlaygroundError.kind === 'rate_limit' || retryPlaygroundError.kind === 'server';
   474→   474→            if (!transient || !nothingStreamed || retryPlaygroundError.kind === 'abort') {
   475→   475→              throw retryError; // permanent failure or user abort — handle below
   476→   476→            }
   477→   477→            if (attempt < MAX_AUTO_RETRIES) {
   478→   478→              toast.info(
   479→   479→                `${providerLabel(conversation.provider)} ${
   480→   480→                  retryPlaygroundError.kind === 'rate_limit' ? 'rate limit' : 'server error'
   481→   481→                } — retrying (attempt ${attempt + 2} of ${MAX_AUTO_RETRIES + 1})…`,
   482→   482→              );
   483→   483→              await new Promise((resolve) => setTimeout(resolve, 1200 * (attempt + 1)));
   484→   484→            } else {
   485→   485→              throw retryError; // retries exhausted — handle below
   486→   486→            }
   487→   487→          }
   488→   488→        }
   489→   489→        if (!outcome) throw new Error('Chat failed without a response.');
   490→   490→        appendAssistant(
   491→   491→          outcome.text || '*(empty response)*',
   492→   492→          {
   493→   493→            responseTimeMs: outcome.responseTimeMs,
   494→   494→            usage: outcome.usage,
   495→   495→          },
   496→   496→          outcome.reasoning,
   497→   497→        );
   498→   498→        setRawDebug({
   499→   499→          request: outcome.rawRequest,
   500→   500→          response: outcome.rawResponse,
   501→   501→          status: outcome.status,
   502→   502→          responseTimeMs: outcome.responseTimeMs,
   503→   503→        });
   504→   504→      } catch (error) {
   505→   505→        const playgroundError = toPlaygroundError(error);
   506→   506→        const partial = streamingTextRef.current;
   507→   507→        const partialReasoning = streamingReasoningRef.current;
   508→   508→        if (playgroundError.kind === 'abort') {
   509→   509→          // "Stop" pressed: keep whatever text already arrived.
   510→   510→          if (partial || partialReasoning) {
   511→   511→            appendAssistant(partial || '*(stopped before any text arrived)*', { stopped: true }, partialReasoning);
   512→   512→          }
   513→   513→          toast.info('Generation stopped.');
   514→   514→        } else {
   515→   515→          if (partial || partialReasoning) {
   516→   516→            appendAssistant(
   517→   517→              partial || '*(error before any text arrived)*',
   518→   518→              { stopped: true, error: playgroundError.message },
   519→   519→              partialReasoning,
   520→   520→            );
   521→   521→          }
   522→   522→          setChatError(playgroundError.message);
   523→   523→          setChatErrorAction(null);
   524→   524→          toast.error(playgroundError.message);
   525→   525→          // One-click recovery for the most common OpenRouter failure modes:
   526→   526→          // no credits (402) or a model ID the key cannot use (400/404). A
   527→   527→          // free model is ALWAYS available on free-tier keys, so offer to
   528→   528→          // switch and re-send in a single click.
   529→   529→          if (
   530→   530→            conversation.provider === 'openrouter' &&
   531→   531→            (playgroundError.status === 402 ||
   532→   532→              playgroundError.status === 404 ||
   533→   533→              playgroundError.status === 400)
   534→   534→          ) {
   535→   535→            const freeModel =
   536→   536→              pickFreeModel(modelsRef.current.openrouter.models) ?? DEFAULT_MODELS.openrouter;
   537→   537→            if (freeModel !== conversation.model) {
   538→   538→              setChatErrorAction({
   539→   539→                label: `Retry with ${freeModel}`,
   540→   540→                run: async () => {
   541→   541→                  updateConversationById(conversationId, { model: freeModel });
   542→   542→                  await runCompletion(conversationId, history);
   543→   543→                },
   544→   544→              });
   545→   545→            }
   546→   546→          }
   547→   547→          // Even failed attempts are useful in the Raw panel.
   548→   548→          if (playgroundError.rawRequest) {
   549→   549→            setRawDebug({
   550→   550→              request: playgroundError.rawRequest,
   551→   551→              response: `// Error: ${playgroundError.message}`,
   552→   552→              status: playgroundError.status,
   553→   553→            });
   554→   554→          }
   555→   555→        }
   556→   556→      } finally {
   557→   557→        abortRef.current = null;
   558→   558→        isStreamingRef.current = false;
   559→   559→        streamingTextRef.current = '';
   560→   560→        streamingReasoningRef.current = '';
   561→   561→        setIsStreaming(false);
   562→   562→        setStreamingConvId(null);
   563→   563→        setStreamingText('');
   564→   564→        setStreamingReasoning('');
   565→   565→      }
   566→   566→    },
   567→   567→    [updateConversationById],
   568→   568→  );
   569→   569→
   570→   570→  const sendMessage = useCallback(
   571→   571→    async (text: string) => {
   572→   572→      const conversation = activeConversationRef.current;
   573→   573→      if (!conversation || isStreamingRef.current) return;
   574→   574→      const userMessage: ChatMessage = {
   575→   575→        id: uid(),
   576→   576→        role: 'user',
   577→   577→        content: text,
   578→   578→        createdAt: Date.now(),
   579→   579→      };
   580→   580→      const history = [...conversation.messages, userMessage];
   581→   581→      // Auto-title the conversation from its first user message.
   582→   582→      const patch: Partial<Conversation> = { messages: history };
   583→   583→      if (conversation.messages.length === 0) patch.title = autoTitle(text);
   584→   584→      updateConversationById(conversation.id, patch);
   585→   585→      await runCompletion(conversation.id, history);
   586→   586→    },
   587→   587→    [runCompletion, updateConversationById],
   588→   588→  );
   589→   589→
   590→   590→  const stopStreaming = useCallback(() => {
   591→   591→    abortRef.current?.abort();
   592→   592→  }, []);
   593→   593→
   594→   594→  const editMessage = useCallback(
   595→   595→    async (messageId: string, newContent: string) => {
   596→   596→      const conversation = activeConversationRef.current;
   597→   597→      if (!conversation || isStreamingRef.current) return;
   598→   598→      const index = conversation.messages.findIndex((m) => m.id === messageId);
   599→   599→      if (index === -1) return;
   600→   600→      // Editing a message truncates everything after it, then re-asks the model.
   601→   601→      const history: ChatMessage[] = [
   602→   602→        ...conversation.messages.slice(0, index),
   603→   603→        { ...conversation.messages[index], content: newContent },
   604→   604→      ];
   605→   605→      updateConversationById(conversation.id, { messages: history });
   606→   606→      await runCompletion(conversation.id, history);
   607→   607→    },
   608→   608→    [runCompletion, updateConversationById],
   609→   609→  );
   610→   610→
   611→   611→  const deleteMessage = useCallback(
   612→   612→    (messageId: string) => {
   613→   613→      const conversation = activeConversationRef.current;
   614→   614→      if (!conversation) return;
   615→   615→      updateConversationById(conversation.id, (c) => ({
   616→   616→        messages: c.messages.filter((m) => m.id !== messageId),
   617→   617→      }));
   618→   618→    },
   619→   619→    [updateConversationById],
   620→   620→  );
   621→   621→
   622→   622→  const regenerate = useCallback(async () => {
   623→   623→    const conversation = activeConversationRef.current;
   624→   624→    if (!conversation || isStreamingRef.current) return;
   625→   625→    // Drop trailing assistant message(s) and ask again.
   626→   626→    let messages = conversation.messages;
   627→   627→    while (messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
   628→   628→      messages = messages.slice(0, -1);
   629→   629→    }
   630→   630→    if (messages.length === 0) return;
   631→   631→    updateConversationById(conversation.id, { messages });
   632→   632→    await runCompletion(conversation.id, messages);
   633→   633→  }, [runCompletion, updateConversationById]);
   634→   634→
   635→   635→  const dismissChatError = useCallback(() => {

   637→   637→    setChatErrorAction(null);

   667→   667→  const loadModels = useCallback(async (provider: ProviderId, force = false) => {
   668→   668→    const cached = modelsRef.current[provider];

   670→   670→

   707→   707→  }, [ready]);

   709→   709→  const testApiKey = useCallback(

   723→   723→        if (result.ok) {

   725→   725→          // A valid key means the models endpoint is usable too — (re)load the
   726→   726→          // list right away so the Model section shows available models

   734→   734→        setKeyTestResults((previous) => ({
   735→   735→          ...previous,
   736→   736→          [provider]: { ok: false, message: playgroundError.message },
   737→   737→        }));

   739→   739→      } finally {
   740→   740→        setTestingKey(null);
   741→   741→      }
   742→   742→    },
